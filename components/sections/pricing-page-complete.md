interface CalculatorState {
  // STEP 1: แพ็กเกจหลัก
  step1: {
    package: 'starter' | 'basic' | 'standard' | 'premium';
    navbar: 'basic' | 'animated' | 'megaMenu';
    footer: 'basic' | 'animated' | 'complex';
  };
  
  // STEP 2: หน้าและ Sections
  step2: {
    pages: {
      small: number;   // จำนวนหน้าเล็ก (2-3 sec)

    };
    extraSections: number;  // จำนวน section เพิ่มเติม
  };
  
  // STEP 3: Animation (สัมพันธ์กับ STEP 2)
  step3: {
    pageAnimations: {
      // แต่ละหน้าจาก STEP 2
      [pageId: string]: 'none' | 'basic' | 'complex';
    };
    sectionAnimation: 'none' | 'basic' | 'complex';
    globalAnimations: {
      scroll: boolean;
      pageTransition: boolean;
      loading: boolean;
      text: boolean;
      parallax: boolean;
      hover: boolean;
      cursor: boolean;
      svgMorph: boolean;
      lottie: boolean;
    };
  };
  
  // STEP 4: บริการเสริม
  step4: {
    seo: boolean;
    bilingual: boolean;
    cms: boolean;
    extendedSupport: boolean;
  };
  
  // STEP 5: สิ่งที่ต้องเตรียม

function calculatePrice(state: CalculatorState): PriceSummary {
  let total = 0;

  } else if (state.step1.navbar === 'megaMenu') {

    large: { none: 1200, basic: 1800, complex: 2500 }
  };

  const animationPrices = {

    loading: 500,

    hover: 300,
